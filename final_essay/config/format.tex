% 页面格式设置
% Page formatting

% 页面编号：首页无页码，正文从第2页开始编页码（页脚居中）
\usepackage{fancyhdr}
\fancyhf{}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0pt}
\pagestyle{fancy}

% 自定义标题页样式：
% 1. 三号黑体居中加粗
% 2. 不显示作者和日期
% 3. 标题页不显示页码
\makeatletter
\renewcommand*\maketitle{%
    \begin{center}%
        {\zihao{3}\heiti\bfseries \@title \par}
        \vskip 1em%
        {\global\let\author\@empty}%
        {\global\let\date\@empty}%
        \thispagestyle{empty}%
    \end{center}%
    \setcounter{footnote}{0}%
}
\makeatother

\def\q{\quad}
